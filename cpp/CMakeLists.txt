cmake_minimum_required(VERSION 3.18)
project(cace_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(cace_runtime_capi SHARED src/c_api.cpp)
target_include_directories(cace_runtime_capi PUBLIC include)
add_executable(cace_runtime_smoke tests/runtime_smoke.cpp)
target_include_directories(cace_runtime_smoke PRIVATE include)
add_executable(cace_rmk1_smoke tests/math/rmk1_smoke.cpp)
target_include_directories(cace_rmk1_smoke PRIVATE include)
add_executable(cace_rmk2_smoke tests/math/rmk2_smoke.cpp)
target_include_directories(cace_rmk2_smoke PRIVATE include)
add_executable(cace_rmk3_smoke tests/math/rmk3_smoke.cpp)
target_include_directories(cace_rmk3_smoke PRIVATE include)
add_executable(cace_rmk4_smoke tests/math/rmk4_smoke.cpp)
target_include_directories(cace_rmk4_smoke PRIVATE include)
add_executable(cace_group_actions_smoke tests/math/group_actions_smoke.cpp)
target_include_directories(cace_group_actions_smoke PRIVATE include)
find_package(pybind11 CONFIG QUIET)
if(pybind11_FOUND)
  pybind11_add_module(_cace_runtime src/bindings.cpp)
  target_include_directories(_cace_runtime PRIVATE include)
endif()

enable_testing()
add_test(NAME cace_runtime_smoke COMMAND cace_runtime_smoke)
add_test(NAME cace_rmk1_smoke COMMAND cace_rmk1_smoke)
add_test(NAME cace_rmk2_smoke COMMAND cace_rmk2_smoke)
add_test(NAME cace_rmk3_smoke COMMAND cace_rmk3_smoke)
add_test(NAME cace_rmk4_smoke COMMAND cace_rmk4_smoke)
add_test(NAME cace_group_actions_smoke COMMAND cace_group_actions_smoke)
